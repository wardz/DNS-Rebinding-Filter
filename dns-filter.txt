!
! ——— IPv4 Addresses
! Regexes are non-strict (loosely matches).
!
! ——— 0.0.0.0/8 (This Network)
/^0(?:\.[0-9]{1,3}){3}$/
!
! ——— 10.0.0.0/8 (Private-Use)
/^10(?:\.[0-9]{1,3}){3}$/
!
! ——— 127.0.0.0/8 (Loopback)
/^127(?:\.[0-9]{1,3}){3}$/
!
! ——— 169.254.0.0/16 (Link-Local)
/^169\.254(?:\.[0-9]{1,3}){2}$/
!
! ——— 172.16.0.0/12 (Private-Use)
/^172\.(?:1[6-9]|2[0-9]|3[0-1])(?:\.[0-9]{1,3}){2}$/
!
! ——— 192.168.0.0/16 (Private-Use)
/^192\.168(?:\.[0-9]{1,3}){2}$/
!
! ——— IPv6 Addresses
! Regexes are non-strict (loosely matches).
!
! ——— ::/128 (Unspecified) and ::1/128 (Loopback)
/^::1?$/
!
! ——— fc00::/7 (Unique-Local) and fe80::/10 (Link-Local)
/^f(?:[cd][0-9a-f]|e[89ab])[0-9a-f]:[0-9a-f:]+$/
!
! ——— Private IPv4-mapped Addresses
/^(?:::ffff:|64:ff9b::)(?:ac1[0-9a-f]|c0a8|a9fe|7f[0-9a-f]{2}|a[0-9a-f]{2}|[0-9a-f]{1,2}):[0-9a-f]{1,4}$/
!
! ——— END
